let products = [];
async function loadCatalog() {
const keyInput = document.getElementById("adminKey").value.trim();
if (!keyInput) {
alert("أدخل مفتاح الإدارة");
return;
}
document.getElementById("status").textContent = "جاري التحقق وتراسل السيرفر...";
try {
// 1. تسجيل الدخول الآمن أولاً عبر JSON API
const loginRes = await fetch("/api/admin/login", {
method: "POST",
headers: { "Content-Type": "application/json" },
body: JSON.stringify({ adminKey: keyInput })
});
const contentType = loginRes.headers.get("content-type");
if (!contentType || !contentType.includes("application/json")) {
throw new Error("استجابة الخادم غير صالحة. تأكد من إعداد السيرفر بشكل صحيح.");
}
const loginData = await loginRes.json();
if (!loginRes.ok || !loginData.ok) {
throw new Error(loginData.error || "مفتاح الإدارة غير صحيح");
}
sessionStorage.setItem("ABESHA_ADMIN_KEY", keyInput);
// 2. جلب المنتجات المتاحة بعد نجاح تسجيل الدخول
document.getElementById("status").textContent = "جاري تحميل الكتالوج والأسعار...";
const prodRes = await fetch("/api/products");
const prodData = await prodRes.json();
if (!prodRes.ok || !prodData.ok) {
throw new Error(prodData.error || "فشل جلب المنتجات");
}
products = prodData.products || [];
document.getElementById("loginBox").style.display = "none";
document.getElementById("search").style.display = "block";
document.getElementById("status").textContent = "تم تحميل " + products.length + " منتج بنجاح.";
renderProducts();
} catch (error) {
sessionStorage.removeItem("ABESHA_ADMIN_KEY");
document.getElementById("status").textContent = "فشل الدخول: " + error.message;
}
}
function renderProducts() {
const search = document.getElementById("search").value.trim().toLowerCase();
const filtered = products.filter(product => {
const name = (product.name || "").toLowerCase();
const id = (product.id || "").toLowerCase();
return name.includes(search) || id.includes(search);
});
document.getElementById("products").innerHTML = filtered.slice(0, 100).map(product => `
<div class="product">
<strong>${escapeHtml(product.name || "")}</strong>
<div class="type">النوع: ${escapeHtml(product.type || "topup")}</div>
<div>ID: ${escapeHtml(product.id || "")}</div>
<div style="margin-top: 8px; font-weight: bold; color: #4CAF50;">
السعر النهائي: ${product.price_sdg ? Number(product.price_sdg).toLocaleString("en-US") + " SDG" : "غير متوفر"}
</div>
<button
onclick="showProductDetails('${escapeHtml(product.id || "")}')"
style="
margin-top:12px;
padding:10px 16px;
border-radius:8px;
border:1px solid #1976ff;
background:#0b3d78;
color:white;
cursor:pointer;
"
>
تفاصيل المنتج
</button>
</div>
`).join("");
if (!filtered.length) {
document.getElementById("products").innerHTML = "<p>لا توجد نتائج.</p>";
}
}
function showProductDetails(productId) {
const product = products.find(p => p.id === productId);
if (!product) return;
let details = "📦 تفاصيل المنتج:\n\n";
details += "اسم المنتج: " + product.name + "\n";
details += "الرقم التعريفي (ID): " + product.id + "\n";
details += "السعر المحسوب: " + (product.price_sdg ? product.price_sdg.toLocaleString("en-US") + " SDG" : "غير محدد") + "\n";
alert(details);
}
function escapeHtml(value) {
return String(value)
.replaceAll("&", "&amp;")
.replaceAll("<", "&lt;")
.replaceAll(">", "&gt;")
.replaceAll('"', "&quot;")
.replaceAll("'", "&#039;");
}
